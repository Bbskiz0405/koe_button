rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // sounds 資料夾：允許所有人讀取，只有登入用戶可寫入
    match /sounds/{fileName} {
      allow read: if true; // 任何人都能播放音檔
      allow write: if request.auth != null; // 只有登入用戶能上傳
    }

    // milktea 資料夾：允許所有人讀取，只有登入用戶可寫入
    match /milktea/{fileName} {
      allow read: if true; // 任何人都能播放音檔
      allow write: if request.auth != null; // 只有登入用戶能上傳
    }
  }
}
